⚙️ Preprocessing RTL

// Camera → ASIC input pipeline // 1. Grayscale (nếu camera BGR) gray = (77*R + 150*G + 29*B) >> 8 // uint8 // 2. Q0.7 encode: shift right 1 q07 = uint8(gray) >> 1 // range [0, 127] // 3. TFLite input quantize // float = int8 × 0.0078125 + (−128 × 0.0078125) // i.e. ASIC sends q07 directly as int8 input inp_i8 = int8(q07) // reinterpret [0,127] as signed // First Conv: x_adj = inp_i8 − inp_zp // = q07 − (−128) = q07 + 128 // range: 0+128 to 127+128 = [128, 255]

🖼️ Tại sao FaceMesh trước khi resize?

Lý do: FaceMesh vẽ các đường landmarks mắt (trắng) lên ảnh grayscale trước khi thu nhỏ về 128×128. Khi thu nhỏ, các đường landmarks được "melt" vào texture của vùng mắt, tạo thêm feature rõ ràng để model phân biệt mắt mở/nhắm. Đây là phần preprocessing quan trọng nhất — bỏ landmarks sẽ làm accuracy giảm đáng kể. RTL engineer cần đảm bảo preprocessing pipeline giống hệt lúc training.

⚙️ FC2 RTL + Output decode

Không cần sigmoid trên ASIC. Vì logit > 0 ↔ sigmoid(logit) > 0.5, chỉ cần so sánh out_i8 > zp_out. Với zp=−2 thì threshold là out_i8 > −2.

⚙️ Requantization Formula

// M_float = S_in × S_weight / S_out (per output channel) // Decompose: M_float = M_int32 × 2^(−total_shift) // Tìm total_shift: n = floor(log2(M_float)) + 31 // so that M_int32 ~ 2^31 M_int32 = round(M_float × 2^n) // int32, MSB near 1 tot_shift= n // range typically 30-45 // Hardware implementation: scaled = $signed(acc) * $signed(M_int32) // 64-bit shifted = $signed(scaled) >>> tot_shift // arithmetic with_zp = shifted + zp_out // add output ZP out_q = clip(with_zp, -128, 127) // saturate int8

Module 1 — MAC Unit (int8 × int9 → int32 accumulate)

// MAC unit: 1 multiply-accumulate step // x_adj = x − zp (9-bit); w = weight (8-bit) module mac_unit ( input wire clk, rst, en, input wire signed [7:0] weight, // int8 weight input wire signed [7:0] x_in, // int8 activation input wire signed [7:0] inp_zp, // zero point (signed) output reg signed [31:0] acc // int32 accumulator ); // x_adj: 9-bit to avoid overflow when zp=−128 → x+128 = [0,255] wire signed [8:0] x_adj = {1'b0, x_in} - {{1{inp_zp[7]}}, inp_zp}; // 9-bit wire signed [16:0] product = $signed(weight) * x_adj; // 17-bit always @(posedge clk) begin if (rst) acc <= 32'sd0; else if (en) acc <= acc + {{15{product[16]}}, product}; // sign-extend to 32-bit end endmodule

Module 2 — Requantizer (int32 → int8, per-channel)

// Requantizer: (acc + bias) × M_int32 >> shift + zp_out → int8 module requantizer #( parameter [31:0] M_INT32 = 32'd765562943, // từ tflite metadata parameter [5:0] TOTAL_SHIFT = 6'd39, // typically 30..45 parameter signed [7:0] OUT_ZP = -8'sd2 // output zero point ) ( input wire signed [31:0] acc_biased, // acc + bias (int32) input wire use_relu, // 1=Conv/FC1, 0=FC2 output wire signed [7:0] out_q // int8 result ); // 64-bit multiply — MUST be int64 to avoid overflow wire signed [63:0] scaled = $signed(acc_biased) * $signed(M_INT32); wire signed [31:0] shifted = $signed(scaled) >>> TOTAL_SHIFT; wire signed [31:0] with_zp = shifted + $signed(OUT_ZP); // Saturate to int8 wire signed [7:0] clipped = (with_zp > 127) ? 8'sd127 : (with_zp < -128) ? -8'sd128 : with_zp[7:0]; // ReLU: if negative → 0 (only for Conv and FC1 layers) assign out_q = (use_relu && clipped[7]) ? 8'd0 : clipped; endmodule

Module 5 — GAP 16×16 (Global Average Pool)

// GAP: sum 16×16=256 pixels, divide by 256 = >>8 module gap_16x16 ( input wire clk, rst, en, last_pixel, input wire signed [7:0] pixel_in, // 1 pixel per cycle output reg signed [7:0] avg_out, // int8 average output reg valid ); // 16×16=256 pixels × max 127 = 32,512 → 15-bit sum, use int32 reg signed [31:0] sum; always @(posedge clk) begin if (rst) begin sum <= 0; valid <= 0; end else if (last_pixel) begin // Final pixel: add then divide by 256 = >>8 avg_out <= $signed(sum + pixel_in) >>> 8; valid <= 1; sum <= 0; end else if (en) begin sum <= sum + pixel_in; valid <= 0; end end endmodule // Instantiate 128× (1 per channel) hoặc time-multiplex 1 module × 128 channels

Module 6 — Drowsiness Classifier (No FPU)

// Không cần sigmoid! Chỉ cần compare int8 vs ZP // Thresholds cho softmax alternatives: // logit = 0 → out_i8 = ZP_OUT = −2 → sigmoid = 0.5 // logit > 0 → out_i8 > −2 → ACTIVE // logit < 0 → out_i8 < −2 → FATIGUE // // Drowsy guard: yêu cầu FATIGUE liên tục N frames module drowsiness_classifier #( parameter signed [7:0] ZP_OUT = -8'sd2, parameter [15:0] GUARD_N = 16'd15 // 15 frames @ 30fps ≈ 0.5s ) ( input wire clk, rst, frame_valid, input wire signed [7:0] out_i8, output reg fatigue_alert, output wire is_fatigue // raw per-frame ); assign is_fatigue = ($signed(out_i8) <= $signed(ZP_OUT)); reg [15:0] cnt; always @(posedge clk) begin if (rst) begin cnt <= 0; fatigue_alert <= 0; end else if (frame_valid) begin if (is_fatigue) begin cnt <= (cnt < GUARD_N) ? cnt + 1 : cnt; fatigue_alert <= (cnt >= GUARD_N - 1); end else begin cnt <= 0; fatigue_alert <= 0; end end end endmodule
